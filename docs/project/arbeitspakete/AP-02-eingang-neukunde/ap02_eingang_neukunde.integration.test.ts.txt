import { KREILE_TENANT_SLUG } from "@/lib/tenant";

(globalThis as typeof globalThis & { AsyncLocalStorage: typeof AsyncLocalStorage }).AsyncLocalStorage = AsyncLocalStorage;
vi.setConfig({ testTimeout: 60_000, hookTimeout: 60_000 });

const DATABASE_URL = process.env.DATABASE_URL;
const EXPECTED_DATABASE_URL = process.env.H18_EXPECTED_DATABASE_URL;
if (!DATABASE_URL || !EXPECTED_DATABASE_URL || DATABASE_URL !== EXPECTED_DATABASE_URL) {
  throw new Error("AP02_LOCAL_REPLAY_REQUIRED: DATABASE_URL must equal H18_EXPECTED_DATABASE_URL");
}
const parsed = new URL(DATABASE_URL);
if (parsed.protocol !== "postgresql:" || parsed.hostname !== "127.0.0.1" || parsed.pathname !== "/postgres" || parsed.username !== "postgres") {
  throw new Error("AP02_LOCAL_REPLAY_REQUIRED: dedicated loopback Postgres expected");
}
if (process.env.SUPABASE_SERVICE_ROLE_KEY) throw new Error("AP02_LOCAL_REPLAY_REQUIRED: service role must be unset");

const sql = postgres(DATABASE_URL, { max: 2, prepare: false });
const originalSecret = process.env.APP_SESSION_SECRET;
process.env.APP_SESSION_SECRET = "ap02-neukunde-local-only";
const RUN = randomUUID().slice(0, 8);

type Fixture = { actorId: string; actorIssuedAt: number };

async function withRequest<T>(cookie: string, work: () => Promise<T>): Promise<T> {
  const { NextRequest } = await import("next/server");
  const { createRequestStoreForAPI } = await import("next/dist/server/async-storage/request-store");
  const { workAsyncStorage } = await import("next/dist/server/app-render/work-async-storage.external");
  const { workUnitAsyncStorage } = await import("next/dist/server/app-render/work-unit-async-storage.external");
  const request = new NextRequest("http://127.0.0.1/ap02-neukunde", { headers: { cookie } });
  const workStore = {
    isStaticGeneration: false,
    page: "/ap02-neukunde",
    route: "/ap02-neukunde",
    afterContext: {},
    previouslyRevalidatedTags: [],
    refreshTagsByCacheKind: new Map(),
    shouldTrackFetchMetrics: false,
    deploymentId: "test",
    buildId: "test",
    incrementalCache: {},
    cacheComponentsEnabled: false,
    runInCleanSnapshot: (fn: (...args: never[]) => unknown, ...args: never[]) => fn(...args),
    reactServerErrorsByDigest: new Map(),
  } as unknown as import("next/dist/server/app-render/work-async-storage.external").WorkStore;
  const requestStore = createRequestStoreForAPI(
    request,
    { pathname: "/ap02-neukunde", search: "" },
    { tags: [], expirationsByCacheKind: new Map() },
    undefined,
    undefined as never,
  );
  return workAsyncStorage.run(workStore, () => workUnitAsyncStorage.run(requestStore, work));
}

async function withSession<T>(fixture: Fixture, work: () => Promise<T>): Promise<T> {
  const { COOKIE_NAME, getSecretKey, signAppSession } = await import("@/lib/server/appSession");
  const session: AppSession = {
    userId: fixture.actorId,
    tenantId: KREILE_TENANT_SLUG,
    role: "meister",
    displayName: "AP02 Neukunde",
    issuedAt: fixture.actorIssuedAt,
    expiresAt: Date.now() + 5 * 60_000,
  };
  return withRequest(`${COOKIE_NAME}=${signAppSession(session, getSecretKey())}`, work);
}

async function createFixture(rights: string[]): Promise<Fixture> {
  const actorId = randomUUID();
  const updatedAt = new Date(Date.now() - 5_000).toISOString();
  await sql.begin(async (transaction) => {
    await transaction`
      INSERT INTO public.app_users
        (id, tenant_id, email, full_name, role, active, created_at, updated_at)
      VALUES (
        ${actorId}::uuid, ${KREILE_TENANT_SLUG}, ${`ap02-neukunde-${actorId}@local.invalid`},
        'AP02 Neukunde', 'meister', true, ${updatedAt}::timestamptz, ${updatedAt}::timestamptz
      )
    `;
    await transaction`
      INSERT INTO public.app_user_rights (tenant_id, user_id, bereich, granted_by)
      SELECT ${KREILE_TENANT_SLUG}, ${actorId}::uuid, area, ${actorId}::uuid
      FROM unnest(${rights}::text[]) area
    `;
  });
  const [clock] = await sql<{ updated_at_ms: string }[]>`
    SELECT floor(extract(epoch FROM updated_at) * 1000)::bigint::text AS updated_at_ms
    FROM public.app_users WHERE id=${actorId}::uuid
  `;
  const actorIssuedAt = Date.now();
  if (!clock || Number(clock.updated_at_ms) >= actorIssuedAt) throw new Error("AP02_SESSION_ORDER_INVALID");
  return { actorId, actorIssuedAt };
}

let eingangActor: Fixture;

beforeAll(async () => {
  const migrations = await sql<{ version: string }[]>`
    SELECT version::text FROM supabase_migrations.schema_migrations ORDER BY version
  `;
  const repositoryMigrations = readdirSync(join(process.cwd(), "supabase", "migrations"))
    .filter((file) => /^\d+_.+\.sql$/.test(file))
    .map((file) => file.slice(0, file.indexOf("_")))
    .sort();
  expect(migrations.map((row) => row.version)).toEqual(repositoryMigrations);
  eingangActor = await createFixture(["eingang", "auftraege", "kunden"]);
});

afterAll(async () => {
  await sql.end({ timeout: 1 });
  if (originalSecret === undefined) delete process.env.APP_SESSION_SECRET;
  else process.env.APP_SESSION_SECRET = originalSecret;
});

async function kundenMitName(name: string) {
  return sql<{ id: string }[]>`
    SELECT id FROM public.customers WHERE tenant_id=${KREILE_TENANT_SLUG} AND name=${name}
  `;
}

describe("AP-02 Neukunde im Eingang: getippt -> neuer Kunde + Auftrag in einem Schritt", () => {
  it("legt GENAU EINEN Kunden ueber createOrderIntake an und haengt intake_item daran", async () => {
    const { erfasseAuftragAusEingangAction } = await import("@/app/actions/eingang.actions");
    const name = `AP02 Neukunde ${RUN} GmbH`;
    const clientKey = randomUUID();
    const result = await withSession(eingangActor, () => erfasseAuftragAusEingangAction(auftragInput(clientKey, neukunde(name))));
    expect(result.code).toBe("OK");
    if (result.code !== "OK") throw new Error(`AP02_CREATE_FAILED:${result.code}:${result.message}`);
    expect(result.replayed).toBe(false);

    const kunden = await kundenMitName(name);
    expect(kunden).toHaveLength(1);

    const items = await sql<{ kunde_id: string; vorgang_art: string; vorgang_id: string; zuordnung: string; kanal: string; quelle: string }[]>`
      SELECT kunde_id, vorgang_art, vorgang_id, zuordnung, kanal, quelle
      FROM public.intake_item WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_key=${clientKey}
    `;
    expect(items).toEqual([{
      kunde_id: kunden[0]!.id, vorgang_art: "auftrag", vorgang_id: result.orderNumber,
      zuordnung: "hand", kanal: "notiz", quelle: "geraet",
    }]);

    const orders = await sql<{ id: string; customer_id: string }[]>`
      SELECT id, customer_id FROM public.orders WHERE tenant_id=${KREILE_TENANT_SLUG} AND order_number=${result.orderNumber}
    `;
    expect(orders).toEqual([{ id: result.orderId, customer_id: kunden[0]!.id }]);

    const receipts = await sql<{ order_number: string }[]>`
      SELECT order_number FROM private.order_intake_receipts
      WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_event_id=${clientKey}::uuid
    `;
    expect(receipts).toEqual([{ order_number: result.orderNumber }]);
  });

  it("ist idempotent: Wiederholung mit gleichem clientKey legt keinen zweiten Kunden an", async () => {
    const { erfasseAuftragAusEingangAction } = await import("@/app/actions/eingang.actions");
    const name = `AP02 Replay ${RUN} GmbH`;
    const clientKey = randomUUID();
    const customer = neukunde(name);
    const first = await withSession(eingangActor, () => erfasseAuftragAusEingangAction(auftragInput(clientKey, customer)));
    if (first.code !== "OK") throw new Error(`AP02_FIRST_FAILED:${first.code}:${first.message}`);
    const second = await withSession(eingangActor, () => erfasseAuftragAusEingangAction(auftragInput(clientKey, customer)));
    if (second.code !== "OK") throw new Error(`AP02_SECOND_FAILED:${second.code}:${second.message}`);
    expect(second.replayed).toBe(true);
    expect(second.orderNumber).toBe(first.orderNumber);
    expect(second.intakeItemId).toBe(first.intakeItemId);
    expect(await kundenMitName(name)).toHaveLength(1);

    const [counts] = await sql<{ items: number; receipts: number }[]>`
      SELECT
        (SELECT count(*)::int FROM public.intake_item WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_key=${clientKey}) AS items,
        (SELECT count(*)::int FROM private.order_intake_receipts WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_event_id=${clientKey}::uuid) AS receipts
    `;
    expect(counts).toEqual({ items: 1, receipts: 1 });
  });

  it("weist einen ungueltigen Neukunden ab und schreibt nichts", async () => {
    const { erfasseAuftragAusEingangAction } = await import("@/app/actions/eingang.actions");
    const name = "X";
    const clientKey = randomUUID();
    const result = await withSession(eingangActor, () => erfasseAuftragAusEingangAction(auftragInput(clientKey, neukunde(name))));
    expect(result.code).toBe("VALIDATION_ERROR");

    const [counts] = await sql<{ items: number; receipts: number }[]>`
      SELECT
        (SELECT count(*)::int FROM public.intake_item WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_key=${clientKey}) AS items,
        (SELECT count(*)::int FROM private.order_intake_receipts WHERE tenant_id=${KREILE_TENANT_SLUG} AND client_event_id=${clientKey}::uuid) AS receipts
    `;
    expect(counts).toEqual({ items: 0, receipts: 0 });
  });
});
